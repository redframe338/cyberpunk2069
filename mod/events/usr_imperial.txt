# 2. Raise the Imperial Standard — THE CIVIL WAR
# Player switches from USR to HRE
country_event = {
	id = usr_imperial.2
	title = usr_imperial.2.t
	desc = usr_imperial.2.d

	picture = GFX_report_event_generic_military_parade

	is_triggered_only = yes

	option = {
		name = usr_imperial.2.a

		# Give HRE cores on its starting territory
		195 = { add_core_of = HRE }
		208 = { add_core_of = HRE }
		213 = { add_core_of = HRE }
		263 = { add_core_of = HRE }
		264 = { add_core_of = HRE }
		248 = { add_core_of = HRE }
		253 = { add_core_of = HRE }
		351 = { add_core_of = HRE }
		214 = { add_core_of = HRE }
		218 = { add_core_of = HRE }
		217 = { add_core_of = HRE }
		238 = { add_core_of = HRE }
		228 = { add_core_of = HRE }
		244 = { add_core_of = HRE }

		# Transfer states to HRE
		# Northwestern Russia (St Petersburg core)
		HRE = { transfer_state = 195 }
		HRE = { transfer_state = 208 }
		HRE = { transfer_state = 213 }
		HRE = { transfer_state = 263 }
		HRE = { transfer_state = 264 }
		HRE = { transfer_state = 244 }
		# Central Russia (Yaroslavl, Ivanovo, Vologda)
		HRE = { transfer_state = 248 }
		HRE = { transfer_state = 253 }
		HRE = { transfer_state = 351 }
		HRE = { transfer_state = 214 }
		# Cossack South (Rostov, Stalingrad, Taganrog)
		HRE = { transfer_state = 218 }
		HRE = { transfer_state = 217 }
		HRE = { transfer_state = 238 }
		HRE = { transfer_state = 228 }

		# Give HRE its starting spirit and army
		HRE = {
			add_ideas = HRE_uncrowned_pretender
			add_manpower = 500000
			load_oob = "HRE_civil_war"
		}

		# HRE declares war on USR (Neo-Soviet Union)
		HRE = {
			declare_war_on = {
				target = USR
				type = annex_everything
			}
		}

		# Switch the player to HRE
		HRE = {
			change_tag_from = USR
		}
	}
}
