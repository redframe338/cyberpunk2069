add_namespace = nus_civil_war

# ================================================================
# NUS CIVIL WAR EVENTS
# One flavor / trigger / victory event trio per ideological path.
#
# Militech (MLT) and Petrochem (PET) allegiance is decided by
# comparing the four path-specific influence variables set and
# grown by each path's focuses. Whichever path has courted a
# corporation hardest wins its backing; if no path has bothered
# (all variables tied at 0, as is typical for the Communist path)
# the corporation defaults to opposing the government.
# ================================================================

# ----------------------------------------------------------------
# PATH 1 — FASCIST UPRISING (America First Movement)
# ----------------------------------------------------------------

# 1. Flavor: The Nation Divides
country_event = {
	id = nus_civil_war.1
	title = nus_civil_war.1.t
	desc = nus_civil_war.1.d
	picture = GFX_report_event_generic_rally
	is_triggered_only = yes

	option = {
		name = nus_civil_war.1.a
		add_stability = -0.02
	}
	option = {
		name = nus_civil_war.1.b
		add_political_power = 25
	}
}

# 2. Trigger: March on Washington
country_event = {
	id = nus_civil_war.2
	title = nus_civil_war.2.t
	desc = nus_civil_war.2.d
	picture = GFX_report_event_generic_military_parade
	is_triggered_only = yes

	option = {
		name = nus_civil_war.2.a
		custom_effect_tooltip = NUS_fascist_civil_war_erupts_tt

		start_civil_war = {
			ideology = neutrality
			size = 0.5
		}
		set_country_flag = NUS_fascist_civil_war_started
		add_stability = -0.10
		add_war_support = 0.10

		# --- Militech allegiance ---
		if = {
			limit = {
				check_variable = { NUS_militech_fascist_influence > NUS_militech_corporate_influence }
				check_variable = { NUS_militech_fascist_influence > NUS_militech_federalist_influence }
				check_variable = { NUS_militech_fascist_influence > NUS_militech_communist_influence }
			}
			custom_effect_tooltip = NUS_militech_backs_movement_tt
			add_ideas = NUS_militech_wartime_support
			add_political_power = 75
		}
		else = {
			custom_effect_tooltip = NUS_militech_backs_loyalists_tt
			add_ideas = NUS_militech_wartime_defection
			add_stability = -0.05
		}

		# --- Petrochem allegiance ---
		if = {
			limit = {
				check_variable = { NUS_petrochem_fascist_influence > NUS_petrochem_corporate_influence }
				check_variable = { NUS_petrochem_fascist_influence > NUS_petrochem_federalist_influence }
				check_variable = { NUS_petrochem_fascist_influence > NUS_petrochem_communist_influence }
			}
			custom_effect_tooltip = NUS_petrochem_backs_movement_tt
			add_ideas = NUS_petrochem_wartime_support
			add_political_power = 75
		}
		else = {
			custom_effect_tooltip = NUS_petrochem_backs_loyalists_tt
			add_ideas = NUS_petrochem_wartime_defection
			add_stability = -0.05
		}
	}
}
