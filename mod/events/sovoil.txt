# A low-influence SovOil receives a permanent loyal state charter while
# remaining a separate playable strategic corporation.
country_event = {
	id = sovoil.11
	title = sovoil.11.t
	desc = sovoil.11.d
	picture = GFX_report_event_generic_sign_treaty1
	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = sovoil.11.a
		add_ideas = SVO_loyal_union_instrument
		set_country_flag = SVO_permanent_union_charter
		set_variable = { sovoil_state_influence = 20 }
		add_stability = 0.05
		USR = {
			add_stability = 0.03
			add_political_power = 50
		}
	}
}
