# ================================================================
# EVENT 11: Government Suspicion — Moscow Investigates
# Random event that fires when suspicion is high
# ================================================================
country_event = {
	id = usr_sovoil_coup.11
	title = usr_sovoil_coup.11.t
	desc = usr_sovoil_coup.11.d
	picture = GFX_report_event_generic_rally

	trigger = {
		tag = USR
		has_idea = USR_board_resolution_authorized
		check_variable = { USR_government_suspicion > 40 }
		NOT = { has_country_flag = USR_moscow_investigated_recent }
		NOT = { has_completed_focus = USR_the_executive_ultimatum }
	}

	mean_time_to_happen = { months = 6 }

	immediate = {
		set_country_flag = { flag = USR_moscow_investigated_recent days = 180 value = 1 }
	}

	# Bribe the investigators
	option = {
		name = usr_sovoil_coup.11.a
		ai_chance = { factor = 50 }
		add_political_power = -75
		add_to_variable = { USR_government_suspicion = -15 }
		custom_effect_tooltip = USR_investigators_bribed_tt
	}

	# Accelerate the coup timeline
	option = {
		name = usr_sovoil_coup.11.b
		ai_chance = { factor = 30 }
		add_to_variable = { USR_coup_readiness = 5 }
		add_to_variable = { USR_government_suspicion = 10 }
		custom_effect_tooltip = USR_accelerate_timeline_tt
	}

	# Sacrifice a minor executive to deflect attention
	option = {
		name = usr_sovoil_coup.11.c
		ai_chance = { factor = 20 }
		add_to_variable = { USR_government_suspicion = -20 }
		add_to_variable = { USR_coup_readiness = -5 }
		add_stability = 0.02
		custom_effect_tooltip = USR_sacrifice_executive_tt
	}
}
