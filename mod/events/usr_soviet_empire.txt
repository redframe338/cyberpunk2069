add_namespace = usr_soviet_empire

# Event 1: The Emergency Session
country_event = {
	id = usr_soviet_empire.1
	title = usr_soviet_empire.1.t
	desc = usr_soviet_empire.1.d
	picture = GFX_report_event_generic_parliament

	is_triggered_only = yes

	option = {
		name = usr_soviet_empire.1.a
		add_stability = -0.05
		add_war_support = 0.10
		add_political_power = -50
		custom_effect_tooltip = USR_emergency_session_tt
	}
}

# Event 2: The Coup Without Gunfire
country_event = {
	id = usr_soviet_empire.2
	title = usr_soviet_empire.2.t
	desc = usr_soviet_empire.2.d
	picture = GFX_report_event_generic_rally2

	is_triggered_only = yes

	option = {
		name = usr_soviet_empire.2.a

		# Remove conspiracy spirit
		remove_ideas = USR_committee_restoration_spirit

		# Set politics — communism stays but with new leadership
		set_politics = {
			ruling_party = communism
			elections_allowed = no
		}

		# Voronin takes power
		set_party_name = {
			ideology = communism
			long_name = USR_soviet_empire_party_long
			name = USR_soviet_empire_party
		}

		# Apply military government spirit
		add_ideas = USR_imperial_presidium_spirit

		# Annex SovOil — absorb its territory
		if = {
			limit = { country_exists = SVO }
			annex_country = { target = SVO transfer_troops = yes }
		}

		# Annex Gazprom if it exists as a country
		if = {
			limit = { country_exists = GAZ }
			annex_country = { target = GAZ transfer_troops = yes }
		}

		# Military gains
		add_manpower = 50000
		army_experience = 25

		custom_effect_tooltip = USR_coup_without_gunfire_tt
	}
}

# Event 3: Proclaim the Soviet Empire
country_event = {
	id = usr_soviet_empire.3
	title = usr_soviet_empire.3.t
	desc = usr_soviet_empire.3.d
	picture = GFX_report_event_generic_rally2

	is_triggered_only = yes

	option = {
		name = usr_soviet_empire.3.a

		# Apply the Soviet Empire cosmetic tag
		set_cosmetic_tag = USR_soviet_empire

		# Remove old spirits, apply capstone
		remove_ideas = USR_imperial_presidium_spirit
		add_ideas = USR_red_colossus_spirit

		# Massive legitimacy event
		add_stability = 0.10
		add_war_support = 0.15
		add_political_power = 150

		# Rename the party
		set_party_name = {
			ideology = communism
			long_name = USR_soviet_empire_final_long
			name = USR_soviet_empire_final
		}

		custom_effect_tooltip = USR_proclaim_soviet_empire_tt
	}
}
