# 3. Operation Blue Dawn — The Coup
# Fired when "Operation Blue Dawn" focus is completed
country_event = {
	id = usr_blue_dawn.3
	title = usr_blue_dawn.3.t
	desc = usr_blue_dawn.3.d

	picture = GFX_report_event_generic_military_parade

	is_triggered_only = yes

	option = {
		name = usr_blue_dawn.3.a
		custom_effect_tooltip = USR_blue_dawn_launched_tt

		# Player becomes the Russian Federation
		set_politics = {
			ruling_party = democratic
			elections_allowed = yes
		}
		set_popularities = {
			communism = 15
			neutrality = 10
			democratic = 70
			fascism = 5
		}
		set_cosmetic_tag = USR_russian_federation

		# Release Gazprom as allied puppet with Western Siberian gas fields
		580 = { add_core_of = GAZ }
		GAZ = {
			transfer_state = 580
			set_politics = {
				ruling_party = democratic
				elections_allowed = yes
			}
			set_popularities = {
				communism = 5
				neutrality = 15
				democratic = 75
				fascism = 5
			}
		}
		set_autonomy = {
			target = GAZ
			autonomous_state = autonomy_puppet
		}

		# Neo-Soviet loyalists rebel — the old guard fights back
		start_civil_war = {
			ideology = communism
			size = 0.3
		}

		# Russian Federation declares war on SovOil
		declare_war_on = {
			target = SVO
			type = annex_everything
		}

		# Gazprom also declares war on SovOil
		GAZ = {
			declare_war_on = {
				target = SVO
				type = annex_everything
			}
		}

		add_stability = -0.10
		add_war_support = 0.15
	}
}
