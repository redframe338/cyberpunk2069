# ================================================================
# EVENT 2: Competing Claims — A New Gas Field
# Random event: SovOil and Gazprom dispute a newly discovered field
# ================================================================
country_event = {
	id = usr_gazprom.2
	title = usr_gazprom.2.t
	desc = usr_gazprom.2.d
	picture = GFX_report_event_generic_factory

	trigger = {
		tag = USR
		has_idea = USR_gas_arteries_of_union
		NOT = { has_country_flag = USR_field_dispute_recent }
	}

	mean_time_to_happen = { months = 24 }

	immediate = {
		set_country_flag = { flag = USR_field_dispute_recent days = 360 value = 1 }
	}

	# Award the field to Gazprom
	option = {
		name = usr_gazprom.2.a
		ai_chance = { factor = 40 }
		add_resource = { type = oil amount = 4 state = 571 }
		add_stability = 0.02
		add_to_variable = { sovoil_state_influence = -5 }
		custom_effect_tooltip = USR_gazprom_field_favor_tt
	}

	# Award the field to SovOil
	option = {
		name = usr_gazprom.2.b
		ai_chance = { factor = 40 }
		add_resource = { type = oil amount = 6 state = 571 }
		add_to_variable = { sovoil_state_influence = 5 }
		add_stability = -0.02
		custom_effect_tooltip = USR_sovoil_field_favor_tt
	}

	# Split the field between them
	option = {
		name = usr_gazprom.2.c
		ai_chance = { factor = 20 }
		add_resource = { type = oil amount = 3 state = 571 }
		add_political_power = 25
	}
}

# ================================================================
# EVENT 7: The Proposed Great Energy Merger
# SovOil and Gazprom — should the government merge them?
# Fires on the SovOil Ascendancy path
# ================================================================
country_event = {
	id = usr_gazprom.7
	title = usr_gazprom.7.t
	desc = usr_gazprom.7.d
	picture = GFX_report_event_generic_factory

	trigger = {
		tag = USR
		has_completed_focus = USR_sovoil_ascendancy
		OR = {
			has_idea = USR_gazprom_privileged_corporation
			has_idea = USR_gazprom_entrenched_bureaucracy
		}
		NOT = { has_country_flag = USR_merger_proposed }
	}

	mean_time_to_happen = { months = 12 }

	immediate = {
		set_country_flag = USR_merger_proposed
	}

	# Accept the merger — SovOil absorbs Gazprom
	option = {
		name = usr_gazprom.7.a
		ai_chance = { factor = 40 }
		if = {
			limit = { has_idea = USR_gazprom_privileged_corporation }
			remove_ideas = USR_gazprom_privileged_corporation
		}
		if = {
			limit = { has_idea = USR_gazprom_entrenched_bureaucracy }
			remove_ideas = USR_gazprom_entrenched_bureaucracy
		}
		add_ideas = USR_gazprom_great_energy_merger
		add_to_variable = { sovoil_state_influence = 20 }
		add_stability = -0.05
		custom_effect_tooltip = USR_merger_accept_tt
	}

	# Reject — preserve Gazprom as a counterweight
	option = {
		name = usr_gazprom.7.b
		ai_chance = { factor = 40 }
		add_to_variable = { sovoil_state_influence = -10 }
		add_political_power = -50
		add_stability = 0.03
		custom_effect_tooltip = USR_merger_reject_tt
	}

	# Delay the decision — political maneuvering
	option = {
		name = usr_gazprom.7.c
		ai_chance = { factor = 20 }
		add_political_power = -25
		clr_country_flag = USR_merger_proposed
		custom_effect_tooltip = USR_merger_delay_tt
	}
}
